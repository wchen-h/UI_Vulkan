# ============================================================
# UI_Vulkan — 首次在新机器上部署步骤
# ============================================================

# ============================================================
# Linux (Ubuntu / Debian)
# ============================================================

# 1. 前置工具
sudo apt update
sudo apt install -y git cmake build-essential
sudo apt install -y libvulkan-dev vulkan-tools   # Vulkan SDK
sudo apt install -y libxrandr-dev libxinerama-dev libxcursor-dev libxi-dev libxext-dev  # GLFW 依赖

# 2. 克隆仓库
git clone -b HDR_SDR_split git@github.com:wchen-h/UI_Vulkan.git
cd UI_Vulkan

# 3. 准备外部依赖
mkdir -p external
cd external
git clone --depth 1 -b docking https://github.com/ocornut/imgui.git imgui
curl -O https://raw.githubusercontent.com/nothings/stb/master/stb_image.h
cd ..

# 4. 构建
cmake -B build -S .
cmake --build build -j$(nproc)

# 5. 运行
./build/UI_Vulkan_SDR    # SDR 设备上运行
./build/UI_Vulkan_HDR    # HDR 设备上运行
# 素材已在 pic/cropped_images/ 中，随仓库克隆

# ============================================================
# Windows (10 / 11)
# ============================================================
# 重要：必须按顺序安装，顺序错误会导致 cmake 报错找不到编译器
# ============================================================

# ── 第 1 步：C++ 编译工具链（最关键，必须第一个装） ──
#   以下二选一，推荐方案 A（轻量，无 IDE）

#   ┌─────────────────────────────────────────────────────────────┐
#   │ 方案 A：Build Tools for Visual Studio 2022（推荐）          │
#   │ 不含 IDE，只有命令行编译工具，下载量约 4 GB                  │
#   │ 下载: https://aka.ms/vs/17/release/vs_BuildTools.exe        │
#   │                                                             │
#   │ 安装时勾选工作负载: "使用 C++ 的桌面开发"                   │
#   │   右侧 "安装详细信息" 确认勾选:                             │
#   │     - MSVC v143 - VS 2022 C++ x64/x86 生成工具              │
#   │     - Windows 10 SDK（或 Windows 11 SDK）                   │
#   │     - 用于 Windows 的 C++ CMake 工具  ← 自带 cmake 和 ninja │
#   └─────────────────────────────────────────────────────────────┘

#   ┌─────────────────────────────────────────────────────────────┐
#   │ 方案 B：Visual Studio 2022 Community（含完整 IDE）          │
#   │ 下载: https://visualstudio.microsoft.com/zh-hans/downloads/ │
#   │ 安装选项同方案 A（工作负载和组件勾选内容完全一样）          │
#   └─────────────────────────────────────────────────────────────┘

#   安装完成后，开始菜单搜索 "x64 Native Tools" 打开开发者终端:
#     x64 Native Tools Command Prompt for VS 2022
#   该终端中 cl.exe、cmake.exe、ninja.exe、nmake.exe 全部自动可用
#
#   验证（在该终端中）:
#     cl
#     cmake --version
#     ninja --version

# ── 第 2 步：Git for Windows ──
#   下载: https://git-scm.com/download/win
#   安装时保持默认选项即可（建议勾选 "Git Bash Here"）
#   验证: git --version

# ── 第 3 步：Vulkan SDK ──
#   下载: https://vulkan.lunarg.com/sdk/home#windows
#   安装时保持默认选项
#   安装完成后 重启所有终端窗口 使环境变量生效
#   验证（在 cmd 或新终端中）: glslc --version

# ── 第 4 步：克隆仓库 ──
#   在 "x64 Native Tools Command Prompt for VS 2022" 或 Git Bash 中:
git clone -b HDR_SDR_split git@github.com:wchen-h/UI_Vulkan.git
cd UI_Vulkan

# ── 第 5 步：准备外部依赖 ──
mkdir external
cd external
git clone --depth 1 -b docking https://github.com/ocornut/imgui.git imgui
#   下载 stb_image.h（在 Git Bash 中）:
curl -O https://raw.githubusercontent.com/nothings/stb/master/stb_image.h
#   或在 PowerShell 中:
#   Invoke-WebRequest -Uri https://raw.githubusercontent.com/nothings/stb/master/stb_image.h -OutFile stb_image.h
cd ..

# ── 第 6 步：构建 ──
#   在 "x64 Native Tools Command Prompt for VS 2022" 中（确保 cl.exe 可用）:
cmake -B build -S . -G "Visual Studio 17 2022" -A x64
cmake --build build --config Release -j

#   或使用 Ninja（VS 自带，更快）:
#   cmake -B build -S . -G Ninja
#   cmake --build build --config Release -j

# ── 第 7 步：运行 ──
build\Release\UI_Vulkan_SDR.exe    # SDR 设备上运行
build\Release\UI_Vulkan_HDR.exe    # HDR 设备上运行
# 素材已在 pic/cropped_images/ 中，随仓库克隆

# ── 常见问题 ──
# Q: cmake 报错 "No CMAKE_CXX_COMPILER could be found"
# A: 必须在 "x64 Native Tools Command Prompt for VS 2022" 中运行 cmake，
#    普通 cmd/PowerShell 没有 cl.exe 环境变量。
#    开始菜单搜索 "x64 Native Tools" 即可找到该终端。
#
# Q: cmake -B build 报错 "generator: NMake" 或找不到 NMake
# A: 没有指定 -G 参数时 cmake 默认用 NMake，但 NMake 只在 VS 开发者终端可用。
#    加 -G "Visual Studio 17 2022" 可避免此问题，或直接在 VS 开发者终端中运行。
#
# Q: glslc 不是内部或外部命令
# A: Vulkan SDK 安装后未重启终端。关闭所有 cmd/PowerShell 窗口重新打开。
